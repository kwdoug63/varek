# VAREK generic sector policy — FINANCE
# ------------------------------------------------------------------------------
# For an autonomous agent in a trading / settlement / reconciliation workflow:
# reading market and position data, staging journal entries, reconciling
# ledgers. The controlling risk is an irreversible money movement or an
# unauthorized touch of signing material.
#
# Posture: DEFAULT-DENY. Deny rules carving holes out of an allowed tree come
# first. File rules match the RESOLVED canonical path (v1.12). Signing keys and
# the immutable journal are denied even though they live under allowed trees.
#
# Replace example paths/hosts with your deployment's real locations.
# ------------------------------------------------------------------------------

# v1.13: `readonly` (= access=ro -O_CREAT -O_TRUNC) makes a rule admit reads
# only. Through v1.12 a path rule admitted every open flag, so the rules marked
# "(read)" below, and the loader rules, also admitted writes by the (root)
# agent. /etc/ld.so.cache is allowed BEFORE `deny path /etc/`: first match
# wins, and in the v1.12 ordering that allow could never fire. Check any
# policy with:  tools/vdp_check <policy> lint
#
# `require warden 1.13` makes an older Warden refuse this file (its parser
# would otherwise ignore `readonly` and admit writes).
require warden 1.13

# --- Denied even under an allowed tree (evaluated first) ----------------------
deny  path /var/lib/ledger/keys/           # settlement / signing keys — never
deny  path /var/lib/ledger/journal/posted/ # posted entries are immutable to the agent
allow path /etc/ld.so.cache readonly            # loader cache: read
deny  path /etc/
deny  path /root/

# --- Files the agent may reach ------------------------------------------------
allow path /var/lib/ledger/journal/draft/  # journal entries staged for a maker-checker step
allow path /var/lib/market/snapshots/ readonly  # market / position data (read)
allow path /var/lib/recon/workpapers/      # reconciliation workpapers
allow path /tmp/varek/                      # scratch workspace
allow path /usr/lib/ readonly                   # loader / shared libraries: read, never write
allow path /lib/ readonly                       # loader / shared libraries: read, never write
allow path /lib64/ readonly                     # loader / shared libraries: read, never write

# --- Programs the agent may launch --------------------------------------------
allow exec /usr/bin/python3

# --- Network the agent may reach (deny-only outbound) -------------------------
allow host 127.0.0.1:5432                  # local ledger / positions database
# No external egress: no order gateway, no wire rail, no market venue is
# reachable by the agent. A money movement requires a human on the far side of
# an un-listed, therefore denied, connection.
