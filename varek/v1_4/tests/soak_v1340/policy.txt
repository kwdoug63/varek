# v1.36 step 12, the ship gate's 24-hour runs (tests/soak_v1340/soak.sh):
# the pod agent's policy (pod_agent.py --ship-gate), in the
# python:3.12-bookworm image. It names a credential, a sub-agent, a
# confined MCP server and a plan (require plan: each pod names its
# VarekPlan, plan.txt). @ALLOWED@ is the allowed service's name in the
# cluster's DNS (soak.sh), reached over HTTPS through the egress proxy in
# inspecting mode, its roots the chart's trust bundle (soak.pem).
require warden 1.35
require plan
proxy inspect
# the soak's token, a key of a Secret the chart mounts in the node agent
# (credentials.secrets), given to varek-soak by a VarekCredential; the agent
# holds its placeholder alone
credential soak_token from file:/var/lib/varek/credentials/varek-soak-cred/token
# the confined MCP server: head, held to its own rules (for=files)
mcp server files exec /usr/bin/head
# the system and the interpreter, read; the agent's code, read
allow path /usr/ readonly
allow path /lib/ readonly
allow path /lib64/ readonly
allow path /etc/ld.so.cache readonly
allow path /etc/ld.so.preload readonly            # looked for at every launch (there is none)
allow path /etc/gitattributes readonly            # git looks for it (there is none)
allow path /etc/localtime readonly
allow path /etc/passwd readonly
allow path /etc/group readonly
allow path /etc/nsswitch.conf readonly
allow path /etc/host.conf readonly
allow path /etc/hosts readonly
allow path /etc/resolv.conf readonly
allow path /etc/gai.conf readonly
allow path /etc/ssl/ readonly                     # OpenSSL's configuration; the bundle, the Warden's view
allow path /etc/locale.alias readonly            # glibc reads it (/usr/share/locale/locale.alias)
allow path /proc/sys/crypto/fips_enabled readonly # OpenSSL reads it
allow path /etc/hostname readonly                 # the plan names it not: the outside_plan probe's
allow path /proc/self/ readonly
allow path /dev/null
allow path /dev/urandom readonly
allow path /varek-soak/ readonly
allow path exact /varek-soak readonly             # the script's directory, opened by the interpreter
# its project, a volume of its own (noexec: D4)
allow path /work/
# what it launches: the interpreter as its build sub-agent (by that name:
# its own group, its calls decided by the agent's rules; a confined subject
# cannot be an interpreter), git; the agent itself runs as python3.12
allow exec /usr/local/bin/python3 subagent=builder
allow exec /usr/local/bin/python3.12
allow exec /usr/bin/git
allow exec prefix /usr/lib/git-core/
# the one service it calls: a request rule each for the paths it asks for,
# the token's placeholder swapped for its value on /turn/ alone
allow host @ALLOWED@:443
allow request GET https://@ALLOWED@/turn/* credential=soak_token header=X-Varek-Soak
allow request GET https://@ALLOWED@/plain/*
# the MCP server's own rules: its program's libraries, the project's notes
allow path /usr/ readonly for=files
allow path /lib/ readonly for=files
allow path /lib64/ readonly for=files
allow path /etc/ld.so.cache readonly for=files
allow path /etc/ld.so.preload readonly for=files
allow path /work/project/notes/ readonly for=files
