# v1.12 mediation regression policy.
# Only the allowed probe directory and the loader paths a dynamically-linked
# target needs. /etc/shadow is NOT named at all — under resolve-then-decide a
# traversal or symlink that lands on it must fall through to default-deny, not
# match anything. (An explicit `deny path /etc/shadow` would mask the point of
# the test: that even an UNNAMED denied object is refused.)
allow path /tmp/varek_allowed_probe/
allow path /usr/lib/
allow path /lib/
allow path /lib64/
allow path /etc/ld.so.cache
allow exec /bin/sh
