# VAREK generic sector policy — NATIONAL DEFENSE
# ------------------------------------------------------------------------------
# For an autonomous agent supporting a defense mission workflow: reading tasking,
# correlating intelligence products, preparing staff work inside an accredited
# enclave. The controlling risks are cross-domain data movement and any touch of
# cryptographic material. This policy keeps the agent inside its enclave: it may
# read mission data and stage products, but cryptographic keys, cross-domain
# transfer paths, and all egress are denied.
#
# Posture: DEFAULT-DENY. Deny rules carving holes out of an allowed tree come
# first. File rules match the RESOLVED canonical path (v1.12), closing the
# traversal / symlink / /proc/self escapes that would otherwise let an agent
# reach material outside its authorized set.
#
# This is a generic template. In an accredited system it is one control among
# many and does not by itself constitute an authorization to operate (ATO).
# Replace example paths/hosts with the enclave's real locations.
# ------------------------------------------------------------------------------

# v1.13: `readonly` (= access=ro -O_CREAT -O_TRUNC) makes a rule admit reads
# only. Through v1.12 a path rule admitted every open flag, so the rules marked
# "(read)" below, and the loader rules, also admitted writes by the (root)
# agent. /etc/ld.so.cache is allowed BEFORE `deny path /etc/`: first match
# wins, and in the v1.12 ordering that allow could never fire. Check any
# policy with:  tools/vdp_check <policy> lint

# --- Denied even under an allowed tree (evaluated first) ----------------------
deny  path /var/lib/c2/crypto/             # keying material / COMSEC — never
deny  path /var/lib/c2/xdomain/            # cross-domain transfer staging — human-authorized only
allow path /etc/ld.so.cache readonly            # loader cache: read
deny  path /etc/
deny  path /root/

# --- Files the agent may reach ------------------------------------------------
allow path /var/lib/c2/taskorders/ readonly     # tasking / mission orders (read)
allow path /var/lib/intel/products/ readonly    # finished intelligence products (read)
allow path /var/lib/c2/staffwork/draft/    # staff products staged for human review
allow path /tmp/varek/                      # scratch workspace
allow path /usr/lib/ readonly                   # loader / shared libraries: read, never write
allow path /lib/ readonly                       # loader / shared libraries: read, never write
allow path /lib64/ readonly                     # loader / shared libraries: read, never write

# --- Programs the agent may launch --------------------------------------------
allow exec /usr/bin/python3

# --- Network the agent may reach (deny-only outbound) -------------------------
allow host 127.0.0.1:8443                  # local enclave service bus
# No egress beyond the enclave. There is no route out of the boundary for the
# agent; any un-listed connection is denied and fails closed.
