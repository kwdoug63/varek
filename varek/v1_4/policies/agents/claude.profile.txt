allow path ${HOME}/.claude/                         # settings, sessions, projects, its sign-in

allow path ${HOME}/.local/state/claude/             # its locks

# the native install (~/.local/bin/claude, a link into versions/)
allow path ${HOME}/.local/share/claude/ readonly
allow exec prefix ${HOME}/.local/share/claude/versions/
# the npm package (npm install -g), under a user prefix or the system's
allow path ${HOME}/.npm-global/lib/node_modules/@anthropic-ai/claude-code/ readonly
allow exec prefix ${HOME}/.npm-global/lib/node_modules/@anthropic-ai/claude-code/
allow exec prefix /usr/lib/node_modules/@anthropic-ai/claude-code/
